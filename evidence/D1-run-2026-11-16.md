+"""slugify(text): turn a text into a URL-safe ASCII slug."""
+
+import re
+import unicodedata
+
+_SEPARATOR_RUN = re.compile(r'[^a-z0-9]+')
+
+
+def slugify(text):
+    if not isinstance(text, str):
+        raise TypeError('slugify() argument must be str, not %s' % type(text).__name__)
+
+    normalized = unicodedata.normalize('NFKD', text)
+    without_marks = ''.join(c for c in normalized if not unicodedata.combining(c))
+    lowered = without_marks.lower()
+    return _SEPARATOR_RUN.sub('-', lowered).strip('-')
